  if (toolkits.length > 2) throw new Error("Free plan: only 2 total connections allowed.");

import type { Request, Response, NextFunction } from "express";
import { supabaseAdmin } from "../services/supabaseAdmin";
declare global {
  namespace Express {
    interface Request {
      auth?: { userId: string; email?: string };
    }
  }
}
export async function requireSupabaseAuth(req: Request, res: Response, next: NextFunction) {
  try {
    const authHeader = req.header("authorization") || "";
    const token = authHeader.startsWith("Bearer ") ? authHeader.slice(7) : null;
    if (!token) return res.status(401).json({ error: "Missing Bearer token" });
    const { data, error } = await supabaseAdmin.auth.getUser(token);
    if (error || !data.user) return res.status(401).json({ error: "Invalid token" });
    req.auth = { userId: data.user.id, email: data.user.email ?? undefined };
    return next();
  } catch (e: any) {
    return res.status(500).json({ error: "Auth middleware failed" });
  }
}

import type { Request, Response, NextFunction } from "express";
import { supabaseAdmin } from "../services/supabaseAdmin";
export type QuotaKind = "run" | "write";
export function enforceQuota(kind: QuotaKind) {
  return async (req: Request, res: Response, next: NextFunction) => {
    const userId = req.auth?.userId;
    const localDate = req.localDate;
    const ent = req.entitlements;
    if (!userId || !localDate || !ent) return res.status(500).json({ error: "Missing quota context" });
    // upsert row
    const up = await supabaseAdmin.from("usage_daily").upsert(
      { user_id: userId, local_date: localDate },
      { onConflict: "user_id,local_date" }
    );
    if (up.error) return res.status(500).json({ error: "Failed to init usage" });
    const { data, error } = await supabaseAdmin
      .from("usage_daily")
      .select("runs_used,writes_used")
      .eq("user_id", userId)
      .eq("local_date", localDate)
      .single();
    if (error) return res.status(500).json({ error: "Failed to load usage" });
    const runsUsed = data.runs_used ?? 0;
    const writesUsed = data.writes_used ?? 0;
    if (kind === "run" && runsUsed >= ent.dailyRunLimit) {
      return res.status(429).json({ error: "Daily run limit reached", local_date: localDate });
    }
    if (kind === "write" && writesUsed >= ent.dailyWriteLimit) {
      return res.status(429).json({ error: "Daily write limit reached", local_date: localDate });
    }
    // increment
    const patch =
      kind === "run"
        ? { runs_used: runsUsed + 1 }
        : { writes_used: writesUsed + 1 };
    const upd = await supabaseAdmin
      .from("usage_daily")
      .update(patch)
      .eq("user_id", userId)
      .eq("local_date", localDate);
    if (upd.error) return res.status(500).json({ error: "Failed to update usage" });
    return next();
  };
}

export type RcComputed = {
  userId: string;          // RevenueCat app_user_id = Supabase user id
  planState: PlanState;
  rcEntitlement: "pro" | "max" | null;
  rcPeriodType: "trial" | "normal" | null;
  status: "ACTIVE" | "EXPIRED" | "CANCELED" | "GRACE" | "PAUSED";
  periodEnd: string | null; // ISO timestamp
};
const RcWebhookSchemaLoose = z.object({
  event: z.record(z.any()).optional(),
  // Some RC versions have top-level "api_version"/"type", etc. Keep loose.
}).passthrough();
function parseIsoOrNull(v: any): string | null {
  if (!v) return null;
  if (typeof v === "string") return v;
  return null;
}
/**
 * Extracts "app_user_id" defensively.
 */
function extractUserId(payload: any): string | null {
  const e = payload?.event ?? payload;
  return (
    e?.app_user_id ??
    e?.user_id ??
    e?.subscriber?.app_user_id ??
    e?.customer_info?.app_user_id ??
    null
  );
}
/**
 * Tries to detect active entitlements + period type.
 * RevenueCat commonly provides customer_info.entitlements where each entitlement has:
 * - expires_date / expiration_at_ms
 * - product_identifier
 * - ownership_type
 * - ...and sometimes "is_active"
 */
function extractEntitlementState(payload: any): {
  hasPro: boolean;
  hasMax: boolean;
  proPeriodType: "trial" | "normal" | null;
  maxPeriodType: "trial" | "normal" | null;
  maxPeriodEnd: string | null;
  proPeriodEnd: string | null;
} {
  const e = payload?.event ?? payload;
  const ci =
    e?.customer_info ??
    e?.subscriber?.original_customer_info ??
    e?.subscriber?.customer_info ??
    e?.customerInfo ??
    null;
  const entitlements = ci?.entitlements ?? ci?.entitlement ?? null;
  const now = Date.now();
  const readEnt = (key: string) => {
    const ent = entitlements?.[key];
    if (!ent) return { active: false, periodType: null as any, periodEnd: null as any };
    // Active inference
    const expiresDate =
      parseIsoOrNull(ent?.expires_date) ||
      parseIsoOrNull(ent?.expiresDate) ||
      null;
    const expiresMs =
      typeof ent?.expiration_at_ms === "number"
        ? ent.expiration_at_ms
        : typeof ent?.expires_date_ms === "number"
          ? ent.expires_date_ms
          : null;
    const active =
      typeof ent?.is_active === "boolean"
        ? ent.is_active
        : expiresMs != null
          ? expiresMs > now
          : expiresDate
            ? Date.parse(expiresDate) > now
            : false;
    // Period type inference (trial/normal)
    // Some payloads include: ent.period_type or ent.ownership_type or event.type hints
    const periodType =
      ent?.period_type === "trial" ? "trial" :
      ent?.periodType === "trial" ? "trial" :
      // fallback: check if it contains "trial" boolean or string markers
      ent?.is_trial === true ? "trial" :
      ent?.isTrial === true ? "trial" :
      null;
    const periodEnd =
      expiresDate ??
      (expiresMs != null ? new Date(expiresMs).toISOString() : null);
    return { active, periodType, periodEnd };
  };
  const pro = readEnt("pro");
  const max = readEnt("max");
  return {
    hasPro: pro.active,
    hasMax: max.active,
    proPeriodType: pro.periodType,
    maxPeriodType: max.periodType,
    proPeriodEnd: pro.periodEnd,
    maxPeriodEnd: max.periodEnd,
  };
}
/**
 * Computes plan + limits from RevenueCat payload. Stores raw payload anyway.
 */
export function computeFromRevenueCatWebhook(rawPayload: unknown): RcComputed {
  const payload = RcWebhookSchemaLoose.parse(rawPayload);
  const userId = extractUserId(payload);
  if (!userId) {
    throw new Error("RevenueCat webhook missing app_user_id");
  }
  const { hasPro, hasMax, proPeriodType, maxPeriodType, proPeriodEnd, maxPeriodEnd } =
    extractEntitlementState(payload);
  // Status inference: if no entitlement active => EXPIRED
  // You can refine with event.type if you want.
  let status: RcComputed["status"] = "EXPIRED";
  if (hasMax || hasPro) status = "ACTIVE";
  // Plan precedence: max > pro > free
  let planState: PlanState = "FREE";
  let rcEntitlement: RcComputed["rcEntitlement"] = null;
  let rcPeriodType: RcComputed["rcPeriodType"] = null;
  let periodEnd: string | null = null;
  if (hasMax) {
    planState = "MAX_ACTIVE";
    rcEntitlement = "max";
    rcPeriodType = maxPeriodType ?? "normal";
    periodEnd = maxPeriodEnd;
  } else if (hasPro) {
    planState = proPeriodType === "trial" ? "PRO_TRIAL" : "PRO_ACTIVE";
    rcEntitlement = "pro";
    rcPeriodType = proPeriodType ?? "normal";
    periodEnd = proPeriodEnd;
  } else {
    planState = "FREE";
    rcEntitlement = null;
    rcPeriodType = null;
    periodEnd = null;
  }
  // If we mark ACTIVE but computed FREE, fix:
  if (planState === "FREE") status = "EXPIRED";
  // Normalize periodType: if PRO_TRIAL but missing, set trial
  if (planState === "PRO_TRIAL") rcPeriodType = "trial";
  return {
    userId,
    planState,
    rcEntitlement,
    rcPeriodType,
    status,
    periodEnd,
  };
}
export function entitlementsRowFromPlan(planState: PlanState) {
  const limits = PLAN_LIMITS[planState];
  return {
    plan_state: planState,
    allowed_toolkits: limits.allowedToolkits,
    max_connections: limits.maxConnections,
    write_enabled: limits.writeEnabled,
    daily_run_limit: limits.dailyRunLimit,
    daily_write_limit: limits.dailyWriteLimit,
  };
}

type ComposioError = {
  status: number;
  message: string;
  details?: any;
  retryAfterSeconds?: number;
};
function asComposioError(status: number, body: any, headers: Headers): ComposioError {
  const retryAfter = headers.get("retry-after");
  const retryAfterSeconds = retryAfter ? Number(retryAfter) : undefined;
  const message =
    body?.error?.message ||
    body?.message ||
    body?.error ||
    `Composio request failed (${status})`;
  return { status, message, details: body, retryAfterSeconds };
}
async function composioFetch<T>(path: string, init: RequestInit): Promise<T> {
  if (!env.COMPOSIO_API_KEY) throw new Error("Missing COMPOSIO_API_KEY");
  const res = await fetch(`${BASE_URL}${path}`, {
    ...init,
    headers: {
      "Content-Type": "application/json",
      Authorization: `Bearer ${env.COMPOSIO_API_KEY}`,
      ...(init.headers || {}),
    },
  });
  const text = await res.text();
  let body: any = null;
  try {
    body = text ? JSON.parse(text) : null;
  } catch {
    body = { raw: text };
  }
  if (!res.ok) {
    throw asComposioError(res.status, body, res.headers);
  }
  return body as T;
}

Android → Express: calls POST /actions/run with { toolkit, action, args }

Express → Composio: calls Composio execute endpoint to run the tool using the user’s connected account

import { resolveToolkit } from "../config/toolkits";
export type RouterOutcome =
  | { type: "CLARIFY"; question: string; choices: { label: string; toolkit: string; connected: boolean }[] }
  | { type: "RESOLVED"; toolkit: string };
const PLATFORM_KEYWORDS: Array<{ toolkit: string; keywords: string[] }> = [
  { toolkit: "slack", keywords: ["slack", "workspace", "channel", "channels", "dm", "dms"] },
  { toolkit: "discord", keywords: ["discord", "server", "guild", "channel"] },
  { toolkit: "whatsapp", keywords: ["whatsapp", "wa", "wapp"] },
  { toolkit: "microsoft_teams", keywords: ["teams", "microsoft teams", "ms teams"] },
  { toolkit: "outlook", keywords: ["outlook"] },
  { toolkit: "gmail", keywords: ["gmail", "email", "emails", "mail", "inbox"] },
];
const GENERIC_MESSAGES_PATTERNS = [
  "latest messages",
  "my latest messages",
  "my messages",
  "recent messages",
  "messages",
  "inbox",
];
function normalize(s: string) {
  return s.trim().toLowerCase();
}
function containsAny(text: string, keywords: string[]) {
  return keywords.some(k => text.includes(k));
}
/**
 * Decide whether "messages" should mean chat or email by default:
 * - If user has any chat toolkits connected+allowed, prefer chat
 * - Else if gmail connected+allowed, use gmail
 * - Else clarify with allowed choices
 */
export function inferTargetToolkit(params: {
  text: string;
  // canonical toolkits only
  allowedToolkits: string[];
  connectedToolkits: string[];
}): RouterOutcome {
  const text = normalize(params.text);
  const allowed = new Set(params.allowedToolkits);
  const connected = new Set(params.connectedToolkits);
  // 1) Explicit platform hints in text => resolve directly (if allowed)
  for (const p of PLATFORM_KEYWORDS) {
    if (containsAny(text, p.keywords)) {
      const canonical = p.toolkit; // already canonical here
      if (allowed.has(canonical)) return { type: "RESOLVED", toolkit: canonical };
      // If user said "slack" but it's not allowed, we still return clarify? Usually upgrade.
      // Let the next layer (policy) return UPGRADE_REQUIRED; keep resolution here.
      return { type: "RESOLVED", toolkit: canonical };
    }
  }
  // 2) Not a "messages" intent => no resolution needed here
  const isGenericMessages = GENERIC_MESSAGES_PATTERNS.some(p => text === p || text.includes(p));
  if (!isGenericMessages) {
    return { type: "RESOLVED", toolkit: "" }; // means "no toolkit inferred"
  }
  // 3) Candidate toolkits (messaging)
  const chatCandidates = ["slack", "whatsapp", "discord", "microsoft_teams"];
  const connectedAllowedChat = chatCandidates.filter(t => allowed.has(t) && connected.has(t));
  // If exactly one connected+allowed chat toolkit exists, auto pick it
  if (connectedAllowedChat.length === 1) {
    return { type: "RESOLVED", toolkit: connectedAllowedChat[0] };
  }
  // If multiple chat toolkits are connected+allowed, clarify
  if (connectedAllowedChat.length > 1) {
    return {
      type: "CLARIFY",
      question: "Which app should I check for your latest messages?",
      choices: connectedAllowedChat.map(t => ({ label: labelFor(t), toolkit: t, connected: true })),
    };
  }
  // No chat connected+allowed. If Gmail is connected+allowed, treat "messages" as emails.
  if (allowed.has("gmail") && connected.has("gmail")) {
    return { type: "RESOLVED", toolkit: "gmail" };
  }
  // Otherwise clarify with allowed messaging-like options (connected first)
  const allowedMessaging = ["gmail", "slack", "whatsapp", "discord", "microsoft_teams", "outlook"].filter(t =>
    allowed.has(t)
  );
  const choices = allowedMessaging
    .map(t => ({ label: labelFor(t), toolkit: t, connected: connected.has(t) }))
    .sort((a, b) => Number(b.connected) - Number(a.connected));
  return {
    type: "CLARIFY",
    question: "Which app should I check for your latest messages?",
    choices,
  };
}
function labelFor(toolkit: string) {
  switch (toolkit) {
    case "gmail":
      return "Gmail (email)";
    case "outlook":
      return "Outlook (email)";
    case "microsoft_teams":
      return "Microsoft Teams";
    default:
      // Title-ish
      return toolkit.charAt(0).toUpperCase() + toolkit.slice(1);
  }
}

import { hasCapability } from "../config/capabilities";
export type ClarifyChoice = { toolkit: string; label: string; connected: boolean };
export type RouterOutcome =
  | { type: "CLARIFY"; question: string; choices: ClarifyChoice[] }
  | { type: "RESOLVED"; toolkit: string };
function labelFor(toolkit: string) {
  // You can improve later with a display-name map.
  return toolkit.replace(/_/g, " ");
}
export function inferLatestMessagesTarget(params: {
  allowedToolkits: string[];     // canonical
  connectedToolkits: string[];   // canonical
}): RouterOutcome {
  const allowed = new Set(params.allowedToolkits);
  const connected = new Set(params.connectedToolkits);
  const connectedAllowed = params.connectedToolkits.filter(t => allowed.has(t));
  // Prefer messaging apps
  const connectedMessaging = connectedAllowed.filter(t => hasCapability(t, "MESSAGING"));
  if (connectedMessaging.length === 1) {
    return { type: "RESOLVED", toolkit: connectedMessaging[0] };
  }
  if (connectedMessaging.length > 1) {
    return {
      type: "CLARIFY",
      question: "Which app should I check for your latest messages?",
      choices: connectedMessaging
        .map(t => ({ toolkit: t, label: labelFor(t), connected: true }))
        .sort((a, b) => a.label.localeCompare(b.label)),
    };
  }
  // Fallback to email if no messaging connected
  const connectedEmail = connectedAllowed.filter(t => hasCapability(t, "EMAIL"));
  if (connectedEmail.length === 1) return { type: "RESOLVED", toolkit: connectedEmail[0] };
  const choices: ClarifyChoice[] = [
    ...connectedEmail.map(t => ({ toolkit: t, label: labelFor(t), connected: true })),
    // Offer other allowed (not connected) messaging/email toolkits as connectable choices (optional)
    ...params.allowedToolkits
      .filter(t => !connected.has(t))
      .filter(t => hasCapability(t, "MESSAGING") || hasCapability(t, "EMAIL"))
      .slice(0, 10) // keep UI sane
      .map(t => ({ toolkit: t, label: labelFor(t), connected: false })),
  ];
  return {
    type: "CLARIFY",
    question: "Which app should I check for your latest messages?",
    choices,
  };
}

Goal
Give your users the same “I type a request → it figures out the right tool(s) → it runs and summarizes” experience you got with “Summarize my emails for today”.

That experience comes from combining:

Tool discovery / planning (NL → which tools + in what order)
Execution (run those tools with the user’s connected accounts)
Guardrails (plan restrictions, connection checks, write confirmation, quotas)
Recommended integration pattern (Production)
A) Your backend hosts the “assistant”
Users send text to your Express API:

POST /assistant/plan (NL → plan)
POST /assistant/execute (execute plan step-by-step)
Or one endpoint POST /assistant/run that internally plans + executes until a stop condition.
B) Use Composio for
Connections: generate connect links (Gmail, Calendar, Slack…)
Tool Router / Discovery: turn NL into a tool plan
Tool execution: run the actual tools (GMAIL_FETCH_EMAILS, etc.)
Exact UX flow you want (mirrors what happened here)
1) User types: “Summarize my emails for today”
Backend does:

Determine allowed toolkits (plan) → Free: Gmail + Calendar only
Check Gmail connected?
If not: return NEEDS_CONNECTION with Gmail connect link
Call Composio Tool Router (planning) with:
text: “Summarize my emails for today”
constraints: allowed toolkits, prefer Gmail
user timezone (for “today”)
Tool Router returns something like:
step(s): GMAIL_FETCH_EMAILS with query for today
optional hydration steps
Backend executes steps using Composio tool execution
Backend returns a formatted “Email summary” card to the app
What you must build (minimal)
1) User ↔ Composio identity mapping
Use Supabase user id as Composio entity_id (best).

Every connect link is created with entity_id = userId
Every tool execution is done with the same entity_id
This is how Composio knows which connected Gmail belongs to which user.

2) Integrations screen
Connect Gmail button → backend asks Composio for Gmail connect link → app opens link
After auth, backend can:
either receive a webhook/callback from Composio
or poll “list connections” to mark it Active
3) Assistant endpoints
POST /assistant/run { text } Returns one of:
OK (result)
CLARIFY (needs platform)
NEEDS_CONNECTION (connect link(s))
UPGRADE_REQUIRED (paywall)
NEEDS_CONFIRMATION (draft + confirm token)
Guardrails (why your backend still matters)
Even with Tool Router:

If user is Free, router must not use Slack/Twitter/etc.
If Gmail is not connected, you must stop and ask user to connect
If router proposes a write tool (send email/post tweet), you must require confirmation
So you pass restrictions at planning time and enforce again at execution time.

What to pass into Tool Router to get good results
For high-quality “today” behavior, include:

timezone (from profiles.timezone)
allowed_toolkits (from entitlements)
connected_toolkits (from composio_accounts)
(Optional) “style”: brief/structured summary
Output formatting (to match the feel you saw)
After execution, your backend should transform raw tool outputs into:

Overview counts
Grouped sections (Unread/Important/By sender)
Action items / next steps
This is a lightweight summarizer step (your backend can use an LLM for that, or Composio may offer summarization helpers depending on your setup).
